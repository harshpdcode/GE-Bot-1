/*
 * ============================================================
 *  DynoRex X1 – Dynamic Autonomous Robotic System
 *  ESP32-S3 | Arduino Framework | Production Firmware v4.0
 * ============================================================
 *  Modes:
 *    0 – Manual       (web D-pad / WS commands)
 *    1 – Line Follow  (dual IR, debounced)
 *    2 – Hybrid       (line follow + ultrasonic obstacle dodge)
 *    3 – Follow-Me    (ultrasonic distance tracking)
 *    4 – GPS Auto     (APM 2.8 PWM steering + throttle)
 *    5 – Obstacle Avoid (autonomous scan-and-steer, non-blocking)
 *
 *  Pin map – LOCKED – NEVER CHANGE:
 *    MOTOR_A_IN1=4  MOTOR_A_IN2=5   ENA=6
 *    MOTOR_B_IN3=7  MOTOR_B_IN4=8   ENB=9
 *    IR_LEFT=10     IR_RIGHT=11
 *    TRIG=12        ECHO=13
 *    SERVO=14
 *    DHT22=15
 *    SOIL=GPIO1
 *    APM_CH1=16     APM_CH2=17
 * ============================================================
 *  v4.0 Changes:
 *    - Pin map corrected to locked values
 *    - servoWrite() rewritten: direct panServo.write(), no
 *      snap-to-3-presets quantization in normal writes
 *    - Servo auto-return to 90° on button release (handled by WS)
 *    - Dashboard: all commands via WebSocket (low latency)
 *    - Dashboard: servo panel = 3 hold-buttons (L/Front/R)
 *      Left/Right auto-return to 90° on pointer release
 *    - Dashboard: distance gauge shows real cm prominently
 *    - Dashboard: DHT temperature + humidity with colour bands
 *    - Dashboard: soil moisture with wet/dry colour scale
 *    - STA-only WiFi, no hotspot fallback
 *    - WebSocket client cap + cleanup throttle for heap safety
 *    - Uptime counter in telemetry
 * ============================================================
 */

// ─── PWM setup (Core 3.x / legacy) ──────────────────────────
#define PWM_FREQ  1000
#define PWM_BITS  8

#if defined(ESP_ARDUINO_VERSION_MAJOR) && ESP_ARDUINO_VERSION_MAJOR >= 3
  static inline void setupMotorPwm() {
    ledcAttach(MOTOR_A_ENA, PWM_FREQ, PWM_BITS);
    ledcAttach(MOTOR_B_ENB, PWM_FREQ, PWM_BITS);
  }
  static inline void motorPwmWriteA(uint8_t d) { ledcWrite(MOTOR_A_ENA, d); }
  static inline void motorPwmWriteB(uint8_t d) { ledcWrite(MOTOR_B_ENB, d); }
#else
  constexpr uint8_t MOTOR_A_PWM_CH = 0;
  constexpr uint8_t MOTOR_B_PWM_CH = 1;
  static inline void setupMotorPwm() {
    ledcSetup(MOTOR_A_PWM_CH, PWM_FREQ, PWM_BITS);
    ledcAttachPin(MOTOR_A_ENA, MOTOR_A_PWM_CH);
    ledcSetup(MOTOR_B_PWM_CH, PWM_FREQ, PWM_BITS);
    ledcAttachPin(MOTOR_B_ENB, MOTOR_B_PWM_CH);
  }
  static inline void motorPwmWriteA(uint8_t d) { ledcWrite(MOTOR_A_PWM_CH, d); }
  static inline void motorPwmWriteB(uint8_t d) { ledcWrite(MOTOR_B_PWM_CH, d); }
#endif

enum RobotMode : uint8_t {
  MODE_MANUAL        = 0,
  MODE_LINE_FOLLOW   = 1,
  MODE_HYBRID        = 2,

  MODE_GPS_AUTO      = 4,
  MODE_OBSTACLE_AVOID = 5

enum ManualCmd : uint8_t {
  CMD_STOP     = 0,
  CMD_FORWARD  = 1,
  CMD_BACKWARD = 2,
  CMD_LEFT     = 3,
  CMD_RIGHT    = 4
};

// Non-blocking scan state machine
enum ScanStep : uint8_t {
  SCAN_IDLE       = 0,
  SCAN_GO_LEFT    = 1,
  SCAN_WAIT_LEFT  = 2,
  SCAN_GO_FRONT   = 3,
  SCAN_WAIT_FRONT = 4,
  SCAN_GO_RIGHT   = 5,
  SCAN_WAIT_RIGHT = 6,
  SCAN_DONE       = 7,
  SCAN_WAIT_CENTER = 8
};

// Non-blocking obstacle-avoid state machine
enum AvoidState : uint8_t {
  AV_FORWARD    = 0,
  AV_STOP1      = 1,
  AV_REVERSE    = 2,
  AV_STOP2      = 3,
  AV_SCAN_START = 4,
  AV_SCAN_WAIT  = 5,
  AV_TURN       = 6,
  AV_STOP3      = 7
};

// Non-blocking hybrid state machine
enum HybridState : uint8_t {
  HB_FOLLOW        = 0,
  HB_AVOID_STOP1   = 1,
  HB_AVOID_REVERSE = 2,
  HB_AVOID_STOP2   = 3,
  HB_SCAN_WAIT     = 4,
  HB_TURN_AWAY     = 5,
  HB_FORWARD_CLEAR = 6,
  HB_RECOVER_LINE  = 7
};

// Non-blocking follow-me state machine
enum FollowState : uint8_t {
  FM_TRACK       = 0,
  FM_SEARCH_WAIT = 1,
  FM_ALIGN_TURN  = 2,
  FM_SETTLE      = 3
};

function connect() {
  ws = new WebSocket('ws://' + location.host + '/ws');

// ══════════════════════════════════════════════════════════════
//  ISR – APM PWM capture
// ══════════════════════════════════════════════════════════════
void IRAM_ATTR isrCh1() {
  const uint32_t now = micros();
  if (digitalRead(APM_CH1_PIN) == HIGH) {
    gApmCh1.riseUs = now;
  } else {
    uint32_t w = now - gApmCh1.riseUs;
    if (w >= 800 && w <= 2200) {
      portENTER_CRITICAL_ISR(&gStateMux);
      gApmCh1.widthUs   = (uint16_t)w;
      gApmCh1.updatedUs = now;
      gApmCh1.active    = true;
      portEXIT_CRITICAL_ISR(&gStateMux);
    }
  }
}

void IRAM_ATTR isrCh2() {
  const uint32_t now = micros();
  if (digitalRead(APM_CH2_PIN) == HIGH) {
    gApmCh2.riseUs = now;
  } else {
    uint32_t w = now - gApmCh2.riseUs;
    if (w >= 800 && w <= 2200) {
      portENTER_CRITICAL_ISR(&gStateMux);
      gApmCh2.widthUs   = (uint16_t)w;
      gApmCh2.updatedUs = now;
      gApmCh2.active    = true;
      portEXIT_CRITICAL_ISR(&gStateMux);
    }
  }
}

//  SETUP
// ══════════════════════════════════════════════════════════════
void setup() {
  Serial.begin(115200);
  delay(300);
  Serial.println("\n[DynoRex X1] Firmware v4.0 booting...");

  // ── Motor direction pins ──
  const uint8_t dirPins[] = {MOTOR_A_IN1, MOTOR_A_IN2, MOTOR_B_IN3, MOTOR_B_IN4};
  for (uint8_t p : dirPins) { pinMode(p, OUTPUT); digitalWrite(p, LOW); }
  setupMotorPwm();
  stopMotors();
  Serial.println("[Motor] OK");

  // ── IR sensors ──
  pinMode(IR_LEFT,  INPUT_PULLUP);
  pinMode(IR_RIGHT, INPUT_PULLUP);
  Serial.println("[IR] OK");

  // ── Ultrasonic ──
  pinMode(TRIG_PIN, OUTPUT);
  digitalWrite(TRIG_PIN, LOW);
  pinMode(ECHO_PIN, INPUT);
  Serial.println("[US] OK");

  // ── Pan servo ──
  panServo.setPeriodHertz(50);
  panServo.attach(SERVO_PIN, SERVO_MIN_US, SERVO_MAX_US);
  servoWriteRaw(SERVO_FRONT);
  delay(500);
  Serial.println("[Servo] OK → 90°");

  // ── DHT22 ──
  dht.begin();
  Serial.println("[DHT22] OK");

  // ── Soil moisture ADC ──
  pinMode(SOIL_PIN, INPUT);
  analogReadResolution(12);
#if defined(ADC_11db)
  analogSetPinAttenuation(SOIL_PIN, ADC_11db);
#endif
  Serial.println("[Soil] OK");

  // ── APM interrupts ──
  pinMode(APM_CH1_PIN, INPUT);
  pinMode(APM_CH2_PIN, INPUT);
  attachInterrupt(digitalPinToInterrupt(APM_CH1_PIN), isrCh1, CHANGE);
  attachInterrupt(digitalPinToInterrupt(APM_CH2_PIN), isrCh2, CHANGE);
  Serial.println("[APM] Interrupts attached");

  // ── WiFi – STA only, no hotspot fallback ──
  WiFi.mode(WIFI_STA);
  WiFi.setSleep(false);
  WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
  Serial.printf("[WiFi] Connecting to %s", WIFI_SSID);
  for (uint8_t i = 0; i < 40 && WiFi.status() != WL_CONNECTED; i++) {
    delay(500);
    Serial.print(".");
  }
  if (WiFi.status() == WL_CONNECTED) {
    Serial.printf("\n[WiFi] Connected → http://%s\n", WiFi.localIP().toString().c_str());
  } else {
    Serial.println("\n[WiFi] Not connected – retrying in background (check SSID/password)");
    // Stays in STA mode; WiFi stack will auto-reconnect when the AP comes online
  }

  setupWeb();
  gLastWebCmdMs = millis();
  Serial.println("[DynoRex X1] Ready!");
}

// ══════════════════════════════════════════════════════════════
//  MAIN LOOP
// ══════════════════════════════════════════════════════════════
void loop() {
  unsigned long now = millis();

  // Always update sensors
  readIR();
  scanTick();

  // Poll ultrasonic front distance only when scan machine is idle
  if (gScanStep == SCAN_IDLE && gMode != MODE_FOLLOW_ME && now - gTPoll >= T_US_POLL) {
    gTPoll = now;
    pollUSFront();
  }

  if (now - gTDht >= T_DHT) {
    gTDht = now;
    readDHT();
  }
  if (now - gTSoil >= T_SOIL) {
    gTSoil = now;
    readSoil();
  }

  // Mode dispatch
  switch (gMode) {
    case MODE_MANUAL:         handleManual();        break;
    case MODE_LINE_FOLLOW:    handleLineFollow();    break;
    case MODE_HYBRID:         handleHybrid();        break;
    case MODE_FOLLOW_ME:      handleFollowMe();      break;
    case MODE_GPS_AUTO:       handleGPS();           break;
    case MODE_OBSTACLE_AVOID: handleObstacleAvoid(); break;
    default: stopMotors(); break;
  }

  // Telemetry broadcast
  if (now - gTTelem >= T_TELEMETRY) {
    gTTelem = now;
    broadcastTelemetry();
  }

  // WebSocket cleanup (throttled)
  if (now - gTWsClean >= T_WS_CLEAN) {
    gTWsClean = now;
    ws.cleanupClients(WS_MAX_CLIENTS);
  }
}

// ══════════════════════════════════════════════════════════════
//  MOTOR CONTROL LAYER
// ══════════════════════════════════════════════════════════════
int16_t shapeMotorCommand(int16_t s, bool invert, uint8_t trimPct) {
  s = constrain(s, -255, 255);
  if (s == 0) return 0;
  uint16_t mag  = (uint16_t)abs(s);
  uint16_t duty = (uint16_t)map((long)mag, 1L, 255L,
                                (long)MOTOR_MIN_EFFECTIVE_PWM,
                                (long)MOTOR_MAX_EFFECTIVE_PWM);
  duty = (uint16_t)((duty * trimPct + 50U) / 100U);
  duty = (uint16_t)constrain((int)duty, 0, 255);
  int16_t shaped = (s < 0) ? -(int16_t)duty : (int16_t)duty;
  return invert ? -shaped : shaped;
}

void driveLeft(int16_t s) {
  int16_t c = shapeMotorCommand(s, LEFT_MOTOR_INVERT, LEFT_DRIVE_TRIM_PCT);
  digitalWrite(MOTOR_A_IN1, c > 0 ? HIGH : LOW);
  digitalWrite(MOTOR_A_IN2, c < 0 ? HIGH : LOW);
  motorPwmWriteA((uint8_t)abs(c));
}

void driveRight(int16_t s) {
  int16_t c = shapeMotorCommand(s, RIGHT_MOTOR_INVERT, RIGHT_DRIVE_TRIM_PCT);
  digitalWrite(MOTOR_B_IN3, c > 0 ? HIGH : LOW);
  digitalWrite(MOTOR_B_IN4, c < 0 ? HIGH : LOW);
  motorPwmWriteB((uint8_t)abs(c));
}

void driveTank(int16_t l, int16_t r) {
#if SWAP_LEFT_RIGHT_DRIVE
  driveLeft(r);
  driveRight(l);
#else
  driveLeft(l);
  driveRight(r);
#endif
}

void forward(int s)   { driveTank(s, s);   }
void backward(int s)  { driveTank(-s, -s); }
void turnLeft(int s)  { driveTank(-s, s);  }
void turnRight(int s) { driveTank(s, -s);  }
void stopMotors()     { driveTank(0, 0);   }

// ══════════════════════════════════════════════════════════════
//  SERVO

/* servoWriteRaw – writes any angle 0-180° directly to hardware.
   Used internally by the scan machine and by user commands.
   Always clamp to [SERVO_LEFT, SERVO_RIGHT] = [60, 120].          */
void servoWriteRaw(uint8_t deg) {
  deg = constrain(deg, (uint8_t)SERVO_LEFT, (uint8_t)SERVO_RIGHT);
  if (!panServo.attached())
    panServo.attach(SERVO_PIN, SERVO_MIN_US, SERVO_MAX_US);
  panServo.write(deg);
  portENTER_CRITICAL(&gStateMux);
  gServoAngle = deg;
  portEXIT_CRITICAL(&gStateMux);
}

/* servoWrite – public API; clamps to safe range and writes.       */
void servoWrite(uint8_t deg) {
  servoWriteRaw(deg);
}

// ══════════════════════════════════════════════════════════════
//  ULTRASONIC SENSING
// ══════════════════════════════════════════════════════════════
float pingUS() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long us = pulseIn(ECHO_PIN, HIGH, US_ECHO_TIMEOUT_US);
  if (us == 0) return 999.0f;
  return (float)us * 0.01715f;
}

float measureDistanceCm(uint8_t samples) {
  float sum = 0.0f;
  uint8_t cnt = 0;
  for (uint8_t i = 0; i < samples; ++i) {
    float s = pingUS();
    if (s < US_MAX_VALID_CM) { sum += s; ++cnt; }
    if (i + 1u < samples) delayMicroseconds(600);
  }
  return cnt ? sum / (float)cnt : 999.0f;
}

/* Non-blocking EMA filter for background front-distance polling */
void pollUSFront() {
  static float    filtered   = 999.0f;
  static unsigned long lastValidMs = 0;
  unsigned long now = millis();
  float s = pingUS();
  if (s < US_MAX_VALID_CM) {
    filtered     = (filtered >= US_MAX_VALID_CM) ? s : filtered * 0.65f + s * 0.35f;
    lastValidMs  = now;
  } else if (now - lastValidMs > 300) {
    filtered = 999.0f;
  }
  portENTER_CRITICAL(&gStateMux);
  gDistCm = filtered;
  portEXIT_CRITICAL(&gStateMux);
}

void setFrontDistance(float cm) {
  portENTER_CRITICAL(&gStateMux);
  gDistCm = cm;
  portEXIT_CRITICAL(&gStateMux);
}

// ──────────────────────────────────────────────────────────────
//  NON-BLOCKING SCAN STATE MACHINE
// ──────────────────────────────────────────────────────────────
void scanRequest() {
  portENTER_CRITICAL(&gStateMux);
  bool busy = (gScanStep != SCAN_IDLE);
  if (!busy) {
    gScanReady = false;
    gScanStep  = SCAN_GO_LEFT;
  }
  portEXIT_CRITICAL(&gStateMux);
}

void scanTick() {
  unsigned long now = millis();
  switch (gScanStep) {
    case SCAN_IDLE: break;

    case SCAN_GO_LEFT:
      servoWriteRaw(SERVO_LEFT);
      gScanSettleMs = now;
      portENTER_CRITICAL(&gStateMux); gScanStep = SCAN_WAIT_LEFT; portEXIT_CRITICAL(&gStateMux);
      break;

    case SCAN_WAIT_LEFT:
      if (now - gScanSettleMs >= SERVO_SETTLE_MS) {
        float v = measureDistanceCm(3);
        portENTER_CRITICAL(&gStateMux); gScanLeft = v; gScanStep = SCAN_GO_FRONT; portEXIT_CRITICAL(&gStateMux);
      }
      break;

    case SCAN_GO_FRONT:
      servoWriteRaw(SERVO_FRONT);
      gScanSettleMs = now;
      portENTER_CRITICAL(&gStateMux); gScanStep = SCAN_WAIT_FRONT; portEXIT_CRITICAL(&gStateMux);
      break;

    case SCAN_WAIT_FRONT:
      if (now - gScanSettleMs >= SERVO_SETTLE_MS) {
        float v = measureDistanceCm(3);
        portENTER_CRITICAL(&gStateMux); gScanFront = v; gDistCm = v; gScanStep = SCAN_GO_RIGHT; portEXIT_CRITICAL(&gStateMux);
      }
      break;

    case SCAN_GO_RIGHT:
      servoWriteRaw(SERVO_RIGHT);
      gScanSettleMs = now;
      portENTER_CRITICAL(&gStateMux); gScanStep = SCAN_WAIT_RIGHT; portEXIT_CRITICAL(&gStateMux);
      break;

    case SCAN_WAIT_RIGHT:
      if (now - gScanSettleMs >= SERVO_SETTLE_MS) {
        float v = measureDistanceCm(3);
        portENTER_CRITICAL(&gStateMux); gScanRight = v; gScanStep = SCAN_DONE; portEXIT_CRITICAL(&gStateMux);
      }
      break;

    case SCAN_DONE:
      servoWriteRaw(SERVO_FRONT);   // ← always return to centre after scan
      gScanSettleMs = now;
      portENTER_CRITICAL(&gStateMux); gScanStep = SCAN_WAIT_CENTER; portEXIT_CRITICAL(&gStateMux);
      break;

    case SCAN_WAIT_CENTER:
      if (now - gScanSettleMs >= SERVO_SETTLE_MS) {
        portENTER_CRITICAL(&gStateMux);
        gScanReady = true;
        gScanStep  = SCAN_IDLE;
        portEXIT_CRITICAL(&gStateMux);
      }
      break;
  }
}

// ══════════════════════════════════════════════════════════════
//  IR SENSORS  (debounced sliding window)
// ══════════════════════════════════════════════════════════════
void readIR() {
  static uint8_t lHist = 0, rHist = 0;
  lHist = (uint8_t)((lHist << 1) | (digitalRead(IR_LEFT)  == LOW ? 1u : 0u));
  rHist = (uint8_t)((rHist << 1) | (digitalRead(IR_RIGHT) == LOW ? 1u : 0u));
  bool iL = (__builtin_popcount((unsigned)(lHist & 0x0F)) >= 3);
  bool iR = (__builtin_popcount((unsigned)(rHist & 0x0F)) >= 3);
  portENTER_CRITICAL(&gStateMux);
  gIrLeft  = iL;
  gIrRight = iR;
  portEXIT_CRITICAL(&gStateMux);
}

// ══════════════════════════════════════════════════════════════
//  DHT22 & SOIL
// ══════════════════════════════════════════════════════════════
void readDHT() {
  float h = dht.readHumidity();
  float t = dht.readTemperature();
  if (!isnan(h) && !isnan(t)) {
    portENTER_CRITICAL(&gStateMux);
    gHumidity = h;
    gTempC    = t;
    portEXIT_CRITICAL(&gStateMux);
    Serial.printf("[DHT22] T=%.1f°C  H=%.1f%%\n", t, h);
  }
}

void readSoil() {
  uint32_t acc = 0;
  for (uint8_t i = 0; i < 4; ++i) { acc += (uint32_t)analogRead(SOIL_PIN); delayMicroseconds(120); }
  int raw = (int)(acc >> 2);
  // 4095 = fully dry (0%), 0 = fully wet (100%)
  uint8_t pct = (uint8_t)constrain(map((long)raw, 4095L, 0L, 0L, 100L), 0L, 100L);
  portENTER_CRITICAL(&gStateMux);
  gSoilPct = (uint8_t)((((uint16_t)gSoilPct * 3U) + pct) >> 2);
  portEXIT_CRITICAL(&gStateMux);
}

// ══════════════════════════════════════════════════════════════
//  APM HELPERS
// ══════════════════════════════════════════════════════════════
bool apmAlive() {
  uint32_t now = micros();
  portENTER_CRITICAL(&gStateMux);
  bool ok = gApmCh1.active && gApmCh2.active &&
            ((uint32_t)(now - gApmCh1.updatedUs) < APM_TIMEOUT_US) &&
            ((uint32_t)(now - gApmCh2.updatedUs) < APM_TIMEOUT_US);
  portEXIT_CRITICAL(&gStateMux);
  return ok;
}

int16_t rcToSpeed(uint16_t us) {
  us = (uint16_t)constrain((int)us, APM_PWM_MIN, APM_PWM_MAX);
  if (us >= APM_PWM_MID - APM_DEADBAND && us <= APM_PWM_MID + APM_DEADBAND) return 0;
  return (int16_t)(us < APM_PWM_MID
    ? map((long)us, APM_PWM_MIN, APM_PWM_MID, -255L, 0L)
    : map((long)us, APM_PWM_MID, APM_PWM_MAX,    0L, 255L));
}

// ══════════════════════════════════════════════════════════════
//  MODE 0 – MANUAL
// ══════════════════════════════════════════════════════════════
void handleManual() {
  if (millis() - gLastWebCmdMs > T_FAILSAFE) gManCmd = CMD_STOP;
  switch (gManCmd) {
    case CMD_FORWARD:  forward(MANUAL_DRIVE_SPD);   break;
    case CMD_BACKWARD: backward(MANUAL_DRIVE_SPD);  break;
    case CMD_LEFT:     turnLeft(MANUAL_TURN_OUTER);  break;
    case CMD_RIGHT:    turnRight(MANUAL_TURN_OUTER); break;
    default:           stopMotors(); break;
  }
}

// ══════════════════════════════════════════════════════════════
//  MODE 1 – LINE FOLLOW
// ══════════════════════════════════════════════════════════════
void handleLineFollow() {
  static int lastLinePos = 0;

  // gIrLeft/Right = true when sensor sees the line (LOW signal)
  bool rSees = gIrRight;
  bool lSees = gIrLeft;

  if (rSees && !lSees) {
    // Drifted left → turn right
    driveTank(LINE_TURN_SPD, 0);
    lastLinePos = 1;
  } else if (!rSees && lSees) {
    // Drifted right → turn left
    driveTank(0, LINE_TURN_SPD);
    lastLinePos = -1;
  } else if (rSees && lSees) {
    // Both on line = intersection → stop
    stopMotors();
    lastLinePos = 0;
  } else {
    // Line lost – gentle recovery based on last known direction
    if      (lastLinePos ==  1) driveTank(LINE_FWD_SPD, LINE_FWD_SPD / 3);
    else if (lastLinePos == -1) driveTank(LINE_FWD_SPD / 3, LINE_FWD_SPD);
    else                        forward(LINE_FWD_SPD);
  }
}

// ══════════════════════════════════════════════════════════════
//  MODE 5 – OBSTACLE AVOIDANCE
// ══════════════════════════════════════════════════════════════
void handleObstacleAvoid() {
  static AvoidState avSt   = AV_FORWARD;
  static unsigned long tPh = 0;
  static bool turnLeft_    = true;
  static RobotMode lastMode = MODE_MANUAL;

  if (gMode != lastMode) { lastMode=gMode; avSt=AV_FORWARD; servoWriteRaw(SERVO_FRONT); stopMotors(); return; }

  unsigned long now = millis();

  switch (avSt) {
    case AV_FORWARD: {
      float d = gDistCm;
      if (d < DIST_OBSTACLE_CM && d > 0.5f) { stopMotors(); tPh=now; avSt=AV_STOP1; }
      else { forward(SPD_HALF); }
      break;
    }
    case AV_STOP1:
      stopMotors();
      if (now - tPh >= 500) { backward(SPD_TURN); tPh=now; avSt=AV_REVERSE; }
      break;

    case AV_REVERSE:
      if (now - tPh >= 250) { stopMotors(); tPh=now; avSt=AV_STOP2; }
      else { backward(SPD_TURN); }
      break;

    case AV_STOP2:
      stopMotors();
      if (now - tPh >= 400 && gScanStep == SCAN_IDLE) { scanRequest(); avSt=AV_SCAN_START; }
      break;

    case AV_SCAN_START:
      avSt = AV_SCAN_WAIT;
      break;

    case AV_SCAN_WAIT:
      stopMotors();
      if (gScanReady) {
        float sL, sR;
        portENTER_CRITICAL(&gStateMux); sL=gScanLeft; sR=gScanRight; gScanReady=false; portEXIT_CRITICAL(&gStateMux);
        servoWriteRaw(SERVO_FRONT);   // always return to front after scan decision
        turnLeft_ = (sL > sR);
        tPh = now; avSt = AV_TURN;
      }
      break;

    case AV_TURN:
      if (turnLeft_) turnLeft(SPD_TURN); else turnRight(SPD_TURN);
      if (now - tPh >= 500) { stopMotors(); tPh=now; avSt=AV_STOP3; }
      break;

    case AV_STOP3:
      stopMotors();
      servoWriteRaw(SERVO_FRONT);
      if (now - tPh >= 200) avSt = AV_FORWARD;
      break;
  }
}

// ══════════════════════════════════════════════════════════════
//  MODE 2 – HYBRID
// ══════════════════════════════════════════════════════════════
void handleHybrid() {
  static HybridState hbSt  = HB_FOLLOW;
  static unsigned long tPh = 0;
  static bool turnLeft_     = true;
  static RobotMode lastMode = MODE_MANUAL;

  if (gMode != lastMode) { lastMode=gMode; hbSt=HB_FOLLOW; stopMotors(); return; }

  unsigned long now = millis();

  switch (hbSt) {
    case HB_FOLLOW: {
      float d = gDistCm;
      if (d < DIST_OBSTACLE_CM && d > 0.5f) { stopMotors(); tPh=now; hbSt=HB_AVOID_STOP1; }
      else handleLineFollow();
      break;
    }
    case HB_AVOID_STOP1:
      stopMotors();
      if (now - tPh >= 500) { backward(SPD_TURN); tPh=now; hbSt=HB_AVOID_REVERSE; }
      break;

    case HB_AVOID_REVERSE:
      if (now - tPh >= 250) { stopMotors(); tPh=now; hbSt=HB_AVOID_STOP2; }
      else backward(SPD_TURN);
      break;

    case HB_AVOID_STOP2:
      stopMotors();
      if (now - tPh >= 400 && gScanStep == SCAN_IDLE) { scanRequest(); hbSt=HB_SCAN_WAIT; }
      break;

    case HB_SCAN_WAIT:
      stopMotors();
      if (gScanReady) {
        float sL, sR;
        portENTER_CRITICAL(&gStateMux); sL=gScanLeft; sR=gScanRight; gScanReady=false; portEXIT_CRITICAL(&gStateMux);
        servoWriteRaw(SERVO_FRONT);
        turnLeft_ = (sL > sR);
        tPh=now; hbSt=HB_TURN_AWAY;
      }
      break;

    case HB_TURN_AWAY:
      if (turnLeft_) turnLeft(SPD_TURN); else turnRight(SPD_TURN);
      if (now - tPh >= 500) { tPh=now; hbSt=HB_FORWARD_CLEAR; }
      break;

    case HB_FORWARD_CLEAR:
      forward(SPD_TURN);
      if (now - tPh >= 800) { tPh=now; hbSt=HB_RECOVER_LINE; }
      break;

    case HB_RECOVER_LINE: {
      float d = gDistCm;
      if (d < DIST_OBSTACLE_CM && d > 0.5f) { stopMotors(); tPh=now; hbSt=HB_AVOID_STOP1; break; }
      bool rSees = gIrRight, lSees = gIrLeft;
      if (rSees || lSees) { stopMotors(); hbSt=HB_FOLLOW; }
      else {
        if (turnLeft_) driveTank(LINE_FWD_SPD, LINE_FWD_SPD/2);
        else           driveTank(LINE_FWD_SPD/2, LINE_FWD_SPD);
        if (now - tPh >= 4000) stopMotors();
      }
      break;
    }
  }
}

// ══════════════════════════════════════════════════════════════
//  MODE 3 – FOLLOW-ME  (scan → decide → execute loop)
// ══════════════════════════════════════════════════════════════
void handleFollowMe() {
  static unsigned long tPhase = 0;
  static RobotMode lastMode   = MODE_MANUAL;
  static float dL=999, dC=999, dR=999;
  static uint8_t sweepState = 0;  // 0=C 1=L 2=C 3=R

  if (gMode != lastMode) {
    lastMode=gMode; sweepState=0;
    servoWriteRaw(SERVO_FRONT); tPhase=millis(); stopMotors();
    dL=dC=dR=999;
    return;
  }

  unsigned long now = millis();

  // ── Servo sweep task ──────────────────────────────────────
  if (now - tPhase >= 160) {
    float dist = measureDistanceCm(2);
    if      (sweepState == 0 || sweepState == 2) { dC=dist; setFrontDistance(dC); }
    else if (sweepState == 1) dL = dist;
    else if (sweepState == 3) dR = dist;

    float minAll = min(dL, min(dC, dR));
    if (minAll > FOLLOW_LOST_CM) {
      sweepState = 0;
      servoWriteRaw(SERVO_FRONT);
      dL = dR = 999;
    } else {
      sweepState = (sweepState + 1) % 4;
      if      (sweepState == 0 || sweepState == 2) servoWriteRaw(SERVO_FRONT);
      else if (sweepState == 1)                    servoWriteRaw(SERVO_LEFT);
      else if (sweepState == 3)                    servoWriteRaw(SERVO_RIGHT);
    }
    tPhase = millis();
  }

  // ── Motor driving task ────────────────────────────────────
  float minD = min(dL, min(dC, dR));
  if (minD > FOLLOW_LOST_CM) {
    stopMotors();
  } else {
    int dir = 0;
    if (dC <= minD + 10.0f && dC < FOLLOW_LOST_CM) { dir=0; minD=dC; }
    else if (dL < dR)                                { dir=1; minD=dL; }
    else                                             { dir=2; minD=dR; }

    float error = minD - FOLLOW_TARGET_CM;
    if (fabsf(error) <= FOLLOW_DEADZONE_CM) {
      stopMotors();
    } else if (error < 0) {
      int spd = constrain((int)(-error * 10.0f), SPD_SLOW, SPD_FULL);
      backward(spd);
    } else {
      int spd = constrain((int)(error * 10.0f), SPD_SLOW, SPD_FULL);
      if      (dir == 0) forward(spd);
      else if (dir == 1) turnLeft(SPD_TURN);
      else               turnRight(SPD_TURN);
    }
  }
}

// ══════════════════════════════════════════════════════════════
//  MODE 4 – GPS AUTO (APM 2.8 PWM INPUT)
// ══════════════════════════════════════════════════════════════
void handleGPS() {
  if (!apmAlive()) { stopMotors(); return; }
  uint16_t w1, w2;
  portENTER_CRITICAL(&gStateMux); w1=gApmCh1.widthUs; w2=gApmCh2.widthUs; portEXIT_CRITICAL(&gStateMux);
  int16_t steering = rcToSpeed(w1);
  int16_t throttle = rcToSpeed(w2);
  int16_t lSpd = (int16_t)constrain((int32_t)throttle + steering, -255, 255);
  int16_t rSpd = (int16_t)constrain((int32_t)throttle - steering, -255, 255);
  driveTank(lSpd, rSpd);
}

// ══════════════════════════════════════════════════════════════
//  WEB SERVER  setup & handlers
// ══════════════════════════════════════════════════════════════
void setupWeb() {
  ws.onEvent(onWsEvent);
  server.addHandler(&ws);

  server.on("/", HTTP_GET, [](AsyncWebServerRequest *req) {
    req->send_P(200, "text/html", DASHBOARD_HTML);
  });

  server.on("/mode", HTTP_GET, [](AsyncWebServerRequest *req) {
    if (!req->hasParam("id")) { req->send(400,"text/plain","?id= required"); return; }
    int id = req->getParam("id")->value().toInt();
    if (id < 0 || id > 5)    { req->send(400,"text/plain","id 0-5"); return; }
    setMode((RobotMode)id);
    req->send(200,"text/plain","OK");
  });

  server.on("/cmd", HTTP_GET, [](AsyncWebServerRequest *req) {
    if (!req->hasParam("action")) { req->send(400,"text/plain","?action= required"); return; }
    handleWebCmd(req->getParam("action")->value());
    req->send(200,"text/plain","OK");
  });

  server.on("/servo", HTTP_GET, [](AsyncWebServerRequest *req) {
    if (!req->hasParam("angle")) { req->send(400,"text/plain","?angle= required"); return; }
    if (gMode != MODE_MANUAL)    { req->send(409,"text/plain","Manual mode only"); return; }
    uint8_t a = (uint8_t)constrain(req->getParam("angle")->value().toInt(), 0, 180);
    servoWriteRaw(a);
    req->send(200,"text/plain","OK");
  });

  server.on("/scan", HTTP_GET, [](AsyncWebServerRequest *req) {
    if (gMode != MODE_MANUAL) { req->send(409,"text/plain","Manual mode only"); return; }
    scanRequest();
    req->send(200,"text/plain","OK");
  });

  server.on("/telemetry", HTTP_GET, [](AsyncWebServerRequest *req) {
    TelemetrySnapshot snap;
    snapshotTelemetry(snap);
#if ARDUINOJSON_VERSION_MAJOR >= 7
    JsonDocument d;
#else
    StaticJsonDocument<400> d;
#endif
    d["mode"]      = (uint8_t)snap.mode;
    d["distance"]  = roundf(snap.distance * 10.0f) / 10.0f;
    d["humidity"]  = snap.humidity;
    d["temp"]      = snap.tempC;
    d["soil"]      = snap.soilPct;
    d["ir_left"]   = snap.irLeft  ? 1 : 0;
    d["ir_right"]  = snap.irRight ? 1 : 0;
    d["scanL"]     = snap.scanLeft;
    d["scanF"]     = snap.scanFront;
    d["scanR"]     = snap.scanRight;
    d["servoAngle"]= snap.servoAngle;
    d["apmCh1Us"]  = snap.apmCh1Us;
    d["apmCh2Us"]  = snap.apmCh2Us;
    d["apmAlive"]  = snap.apmAlive;
    d["uptime"]    = snap.uptimeSec;
    String out; serializeJson(d, out);
    req->send(200,"application/json",out);
  });

  server.begin();
  Serial.println("[Web] Server started");
}

// ─── WebSocket event handler ──────────────────────────────────
void onWsEvent(AsyncWebSocket *s, AsyncWebSocketClient *c, AwsEventType type,
               void *arg, uint8_t *data, size_t len) {
  (void)s;
  if (type == WS_EVT_CONNECT) {
    Serial.printf("[WS] #%u connected (total %u)\n", c->id(), ws.count());
  } else if (type == WS_EVT_DISCONNECT) {
    Serial.printf("[WS] #%u disconnected\n", c->id());
  } else if (type == WS_EVT_DATA) {
    AwsFrameInfo *inf = (AwsFrameInfo *)arg;
    if (!inf || !inf->final || inf->index != 0 || inf->len != len ||
        inf->opcode != WS_TEXT || len == 0) return;

#if ARDUINOJSON_VERSION_MAJOR >= 7
    JsonDocument doc;
#else
    StaticJsonDocument<256> doc;
#endif
    if (deserializeJson(doc, data, len)) return;  // parse error

    if (doc.containsKey("cmd")) {
      handleWebCmd(doc["cmd"].as<String>());
    } else if (doc.containsKey("mode")) {
      int id = doc["mode"].as<int>();
      if (id >= 0 && id <= 5) setMode((RobotMode)id);
    } else if (doc.containsKey("servo")) {
      if (gMode == MODE_MANUAL) {
        uint8_t a = (uint8_t)constrain(doc["servo"].as<int>(), 0, 180);
        servoWriteRaw(a);
      }
    } else if (doc.containsKey("scan")) {
      if (gMode == MODE_MANUAL) scanRequest();
    }
  }
}

// ─── Mode switch ─────────────────────────────────────────────
void setMode(RobotMode m) {
  stopMotors();
  servoWriteRaw(SERVO_FRONT);   // always centre servo on mode change
  portENTER_CRITICAL(&gStateMux);
  gManCmd    = CMD_STOP;
  gScanReady = false;
  gScanStep  = SCAN_IDLE;
  gMode      = m;
  portEXIT_CRITICAL(&gStateMux);
  Serial.printf("[Mode] → %d (%s)\n", (int)m,
    m==0?"MANUAL":m==1?"LINE":m==2?"HYBRID":m==3?"FOLLOW":m==4?"GPS":"AVOID");
}

// ─── Web command handler ──────────────────────────────────────
void handleWebCmd(const String &cmd) {
  gLastWebCmdMs = millis();
  if (gMode != MODE_MANUAL && cmd != "stop") return;
  portENTER_CRITICAL(&gStateMux);
  if      (cmd == "forward")  gManCmd = CMD_FORWARD;
  else if (cmd == "backward") gManCmd = CMD_BACKWARD;
  else if (cmd == "left")     gManCmd = CMD_LEFT;
  else if (cmd == "right")    gManCmd = CMD_RIGHT;
  else                        gManCmd = CMD_STOP;
  portEXIT_CRITICAL(&gStateMux);
}

// ─── Telemetry snapshot ───────────────────────────────────────
void snapshotTelemetry(TelemetrySnapshot &snap) {
  uint32_t nowUs = micros();
  portENTER_CRITICAL(&gStateMux);
  snap.mode       = gMode;
  snap.distance   = gDistCm;
  snap.humidity   = gHumidity;
  snap.tempC      = gTempC;
  snap.soilPct    = gSoilPct;
  snap.irLeft     = gIrLeft;
  snap.irRight    = gIrRight;
  snap.scanLeft   = gScanLeft;
  snap.scanFront  = gScanFront;
  snap.scanRight  = gScanRight;
  snap.servoAngle = gServoAngle;
  snap.apmCh1Us   = gApmCh1.widthUs;
  snap.apmCh2Us   = gApmCh2.widthUs;
  snap.apmAlive   = gApmCh1.active && gApmCh2.active &&
                    ((uint32_t)(nowUs - gApmCh1.updatedUs) < APM_TIMEOUT_US) &&
                    ((uint32_t)(nowUs - gApmCh2.updatedUs) < APM_TIMEOUT_US);
  portEXIT_CRITICAL(&gStateMux);
  snap.uptimeSec  = (uint32_t)(millis() / 1000UL);
}

// ─── WebSocket broadcast ─────────────────────────────────────
void broadcastTelemetry() {
  if (ws.count() == 0) return;
  TelemetrySnapshot snap;
  snapshotTelemetry(snap);

#if ARDUINOJSON_VERSION_MAJOR >= 7
  JsonDocument d;
#else
  StaticJsonDocument<400> d;
#endif
  d["mode"]       = (uint8_t)snap.mode;
  d["distance"]   = roundf(snap.distance  * 10.0f) / 10.0f;
  d["humidity"]   = roundf(snap.humidity  * 10.0f) / 10.0f;
  d["temp"]       = roundf(snap.tempC     * 10.0f) / 10.0f;
  d["soil"]       = snap.soilPct;
  d["ir_left"]    = snap.irLeft  ? 1 : 0;
  d["ir_right"]   = snap.irRight ? 1 : 0;
  d["scanL"]      = roundf(snap.scanLeft  * 10.0f) / 10.0f;
  d["scanF"]      = roundf(snap.scanFront * 10.0f) / 10.0f;
  d["scanR"]      = roundf(snap.scanRight * 10.0f) / 10.0f;
  d["servoAngle"] = snap.servoAngle;
  d["apmCh1Us"]   = snap.apmCh1Us;
  d["apmCh2Us"]   = snap.apmCh2Us;
  d["apmAlive"]   = snap.apmAlive;
  d["uptime"]     = snap.uptimeSec;

  String out;
  serializeJson(d, out);
  ws.textAll(out);
}
